# AI 虚拟主播 —— 依赖清单
#
# 装法（在 vtuber/ 目录下）：
#   python -m venv .venv
#   .venv\Scripts\python.exe -m pip install -r requirements.txt
#
# 或者直接复用一个已经有这些包的 venv 也行。

# ── 三脑 + 网络 ───────────────────────────────────────
httpx==0.28.1           # LLM 调用（DeepSeek 流式）
websockets==17.2        # 火山 TTS 双向流式 + B站弹幕
brotli==1.2.0           # B站弹幕解压（没它也能跑，会自动降级成 zlib）

# ── 声音 ──────────────────────────────────────────────
pygame-ce==2.5.8        # 开窗 + 放声音（pygame 的社区维护版，装法一样）

# ── 画面 ──────────────────────────────────────────────
live2d-py==1.0.0        # Live2D 引擎（Python 绑定）
PyOpenGL==3.1.10        # OpenGL（live2d 渲染要用）
numpy==2.5.3            # 数字数组（单声道→立体声、波形分析）

# ── 嘴型翻译 ──────────────────────────────────────────
pypinyin==0.55.0        # 汉字转拼音（"好" → "hao"）

# ── Web 控制台 ────────────────────────────────────────
fastapi==0.143.1        # 网页接口层（把 Python 函数变成网页能调的接口）
uvicorn==0.54.0         # ASGI 服务器（建事件循环、跑 FastAPI 的那个"服务器"）
